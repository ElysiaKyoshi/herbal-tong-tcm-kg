# 微调依赖（doc09）—— 与主项目分开的独立环境
# 重要：新版 LLaMA-Factory 要求 Python >= 3.11；6GB 显存下请用 0.5B 基座（见 README）
#
# 1) 建独立环境（示例用 venv；不要装进主项目环境）
#    "C:\path\to\python312\python.exe" -m venv D:\venvs\tune312
#    D:\venvs\tune312\Scripts\python -m pip install --upgrade pip
#
# 2) 装 CUDA 版 torch（走 pytorch 官方源更快；cu126 对应 sm_86 的 RTX 30 系）
#    pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu126
#    （本机实测：torch 2.14.0+cu126 / torchvision 0.29.0+cu126 / torchaudio 2.11.0+cu126）
#
# 3) 装 LLaMA-Factory（其 pyproject 会带上下列依赖；国内建议换镜像）
#    git clone --depth 1 https://github.com/hiyouga/LLaMA-Factory.git
#    pip install -e LLaMA-Factory -i https://repo.huaweicloud.com/repository/pypi/simple/
#
# 4) 4-bit QLoRA 必需
#    pip install bitsandbytes -i https://repo.huaweicloud.com/repository/pypi/simple/
