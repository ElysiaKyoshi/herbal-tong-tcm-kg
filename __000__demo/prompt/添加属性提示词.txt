def create_entity(label: str, name: str):
    """创建实体（节点），用 MERGE 保证不重复"""
    if not name or str(name).strip() == "":
        return
    query = f"MERGE (n:{label} {{name: $name}})"
    neo4j_client.run_cypher(query, {"name": name.strip()})


def import_json1(data_list):
    """
    处理第一类 JSON 格式
    {
        "name": "麻黄汤",
        "effects": ["发汗解表", "宣肺平喘"],
        "indications": [{"content": "外感风寒", "type": "疾病"}, ...],
        "ingredients": [{"herb": "麻黄", "amount": "9克"}, ...]
    }
    """
    for item in data_list:
        # 方剂
        prescription_name = item.get("name")
        if prescription_name:
            create_entity("Prescription", prescription_name)

        # 功效
        for eff in item.get("effects", []):
            create_entity("Effect", eff)

        # 疾病 & 症状
        for ind in item.get("indications", []):
            if ind.get("type") == "疾病":
                create_entity("Disease", ind.get("content"))
            elif ind.get("type") == "症状":
                create_entity("Symptom", ind.get("content"))

        # 中药成分
        for ing in item.get("ingredients", []):
            herb = ing.get("herb")
            if herb:
                create_entity("Herb", herb)


def import_json2(data_list):
    """
    处理第二类 JSON 格式
    {
        "名称": "麻黄汤",
        "组成": "麻黄9克、桂枝6克、杏仁6克、炙甘草3克。",
        "功效": "发汗解表，宣肺平喘。",
        "功效分类": "辛温解表",
        "方剂分类": "解表剂",
        "出处标准化": "《伤寒论》"
    }
    """
    for item in data_list:
        # 方剂
        prescription_name = item.get("名称")
        if prescription_name:
            create_entity("Prescription", prescription_name)

        # 功效分类
        eff_cat = item.get("功效分类")
        if eff_cat:
            create_entity("EffectCategory", eff_cat)

        # 方剂分类
        pres_cat = item.get("方剂分类")
        if pres_cat:
            create_entity("PrescriptionCategory", pres_cat)

        # 出处标准化
        source = item.get("出处标准化")
        if source:
            create_entity("Source", source)
