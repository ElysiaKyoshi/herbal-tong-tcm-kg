from langchain_core.output_parsers import JsonOutputParser
from langchain_openai import ChatOpenAI
from pydantic import BaseModel, Field
from typing import List, Optional, Dict, Any, Literal
from langchain_core.prompts import PromptTemplate


# ======================
# 初始化 LLM
# ======================
llm = ChatOpenAI(
    api_key="YOUR_API_KEY_HERE",
    model="deepseek-chat",
    base_url="https://api.deepseek.com"
)

# ======================
# 定义枚举字段
# ======================

# 实体类型枚举
EntityType = Literal[
    "Symptom",   # 症状
    "Disease",   # 疾病
    "Formula",   # 方剂
    "Herb",      # 药材
    "Effect",    # 功效
    "Source"     # 出处
]

# 关系类型枚举
RelationType = Literal[
    "TREATS_DISEASE",      # 治疗疾病（方剂或药材 -> 疾病）
    "ALLEVIATES_SYMPTOM",  # 缓解症状（方剂或药材 -> 症状）
    "HAS_EFFECT",          # 具有功效（方剂或药材 -> 功效）
    "HAS_INGREDIENT",      # 含有中药材（方剂 -> 药材）
    "HAS_SYMPTOM",         # 包含症状（疾病 -> 症状）
    "FROM_SOURCE"          # 来自出处（方剂 -> 出处）
]

# ======================
# 实体属性结构
# ======================

class FormulaAttributes(BaseModel):
    alias: Optional[str] = Field(default=None, description="方剂别名")
    domain: Optional[str] = Field(default="TCM", description="数据来源领域")
    effect: Optional[str] = Field(default=None, description="主要功效")
    indication: Optional[str] = Field(default=None, description="主治病证")
    taboo: Optional[str] = Field(default=None, description="禁忌")
    usage: Optional[str] = Field(default=None, description="服用方法")

class HerbAttributes(BaseModel):
    dosage: Optional[str] = Field(default=None, description="推荐用量")
    effect: Optional[str] = Field(default=None, description="主要功效")
    indication: Optional[str] = Field(default=None, description="主治病证")
    meridian: Optional[str] = Field(default=None, description="归经")
    origin: Optional[str] = Field(default=None, description="来源")
    place: Optional[str] = Field(default=None, description="产地")
    processing: Optional[str] = Field(default=None, description="炮制方法")
    property_flavor: Optional[str] = Field(default=None, description="性味")
    taboo: Optional[str] = Field(default=None, description="禁忌")
    traits: Optional[str] = Field(default=None, description="外形特征")

# ======================
# 实体定义（带枚举类型）
# ======================
class Entity(BaseModel):
    name: str = Field(description="实体名称")
    type: EntityType = Field(description="实体类型（限定值）")
    attributes: Optional[Dict[str, Any]] = Field(
        default=None, description="实体属性"
    )

# ======================
# 关系定义（带枚举类型）
# ======================
class RelationAttributes(BaseModel):
    amount: Optional[str] = Field(default=None, description="方剂中药材用量")

class Relation(BaseModel):
    subject: str = Field(description="关系的起点实体名称")
    relation: RelationType = Field(description="关系类型（限定值）")
    object: str = Field(description="关系的终点实体名称")
    attributes: Optional[RelationAttributes] = Field(default=None, description="关系属性")

# ======================
# 知识图谱结构
# ======================
class TCMKnowledgeGraph(BaseModel):
    entities: List[Entity] = Field(description="抽取的实体列表")
    relations: List[Relation] = Field(description="抽取的关系列表")

# ======================
# 初始化解析器
# ======================
parser = JsonOutputParser(pydantic_object=TCMKnowledgeGraph)

# ======================
# Prompt 模板
# ======================
prompt = PromptTemplate(
    template=(
        "你是一个中医知识图谱抽取专家。请从以下文本中提取结构化知识：\n"
        "包含实体（类型限定为 Symptom, Disease, Formula, Herb, Effect, Source）\n"
        "以及关系（限定为 TREATS_DISEASE, ALLEVIATES_SYMPTOM, HAS_EFFECT, HAS_INGREDIENT, HAS_SYMPTOM, FROM_SOURCE）。\n\n"
        "若文本涉及方剂或药材，请补充对应的属性字段（如功效、性味、剂量等）。\n"
        "所有输出必须符合以下 JSON 格式：\n"
        "{format_instructions}\n\n"
        "输入文本：{text}"
    ),
    input_variables=["text"],
    partial_variables={"format_instructions": parser.get_format_instructions()}
)

# ======================
# 构建抽取链
# ======================
chain = prompt | llm | parser

# ======================
# 示例调用
# ======================
example_text = (
    "葛根汤出自《伤寒论》，具有解肌发表、升阳止泻的功效。"
    "主治外感风寒引起的发热、项背强痛。"
    "方中含有葛根9克、麻黄6克、桂枝6克、芍药6克、生姜9克、大枣4枚、甘草6克。"
)

result = chain.invoke({"text": example_text})
print(result)
